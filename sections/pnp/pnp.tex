\section{Optimization methods for inverse problems}

\subsection{Majorization-Minimization algorithms}

\acrfull{mm} is a family of optimization algorithms that iteratively minimizes a surrogate function that majorizes the original objective function.
The surrogate function is called a \textit{tangent-majorant} and is constructed such that it is easier to minimize than the original objective function.

\begin{definition}
    A \textit{tangent-majorant} of a function differentiable function$f : \mathbb{R}^n \to \mathbb{R}$ at a point $x_0$ is a function $F$ such that:
    \begin{enumerate}
        \item $\forall x \in \mathbb{R}^n, f(x) = F(x,x)$
        \item $\forall x \in \mathbb{R}^n, f(x) \leq F(x,x_0)$
    \end{enumerate}
\end{definition}
\noindent The \acrshort{mm} algorithm consists in iteratively minimizing the tangent-majorant of the objective function at the current iterate. The next iterate is then obtained by minimizing the tangent-majorant, and the process is repeated until convergence :
\begin{algorithm}
    \caption{Majorization-Minimization Algorithm}
    \begin{algorithmic}[1]
        \State Initialize $x_0$
        \For{$k = 0, 1, 2, \ldots$}
            \State Compute the tangent-majorant $F(\cdot, x_k)$
            \State $x_{k+1} = \argmin_x F(x, x_k)$
        \EndFor
    \end{algorithmic}
\end{algorithm}

\subsubsection{Expectation-minimization algorithms}

Let $\theta \in \mathbb{R}^n$ be the probability distribution parameters to be estimated. Let $Y \in \mathbb{R}^m$ be the random variable representing the observed data and $Z \in \mathbb{R}^{m \times n}$ be the random variable representing the unobserved or latent data.
$Z$ is an auxiliary variable that is introduced to simplify the optimization problem and the joint density $p_{Y, Z}(y, z|\theta)$ is called the \textit{complete-data likelihood}.
Thus, the \textit{observed-data likelihood} $p_{Y}(y|\theta)$ is obtained by marginalizing over the latent variable $Z$:
\begin{equation}
    p_{Y}(y|\theta) = \int p_{Y, Z}(y, z|\theta) dz
\end{equation}
The \acrfull{em} framework is a specific instance of the \acrshort{mm} algorithm as the tangent-majorant $Q$ is constructed using the expectation of the complete-data log-likelihood \cite{Lange2016,Dempster1977} :
\begin{align}
    Q(\theta, \theta_k) &= \mathbb{E}_{Z|Y, \theta_k}[\log p_{Y, Z}(Y, Z|\theta)] \\
    &= \int_{\mathbb{R}^{m \times n}} \log p_{Y, Z}(Y, z|\theta) p_{Z|Y, \theta_k}(z|Y, \theta_k) dz
\end{align}
where $\theta_k$ is the current estimate of the parameters and $\theta$ is the variable to be optimized.
The \acrshort{em} algorithm consists of two steps: the E-step, where the expectation of the complete-data log-likelihood is computed, and the M-step, where the parameters are updated by maximizing the tangent-majorant $Q$:
\begin{algorithm}
    \caption{Expectation-Maximization Algorithm}
    \begin{algorithmic}[1]
        \State Initialize $\theta_0$
        \For{$k = 0, 1, 2, \ldots$}
            \State E-step: Compute $Q(\theta, \theta_k)$
            \State M-step: $\theta_{k+1} = \argmax_\theta Q(\theta, \theta_k)$
        \EndFor
    \end{algorithmic}
\end{algorithm}

In the specific setting of Poisson noise, the \acrshort{em} algorithm is known as the \acrfull{mlem} algorithm and is used to estimate the parameters of a Poisson distribution.
The \acrshort{mlem} algorithm is particularly useful in image reconstruction problems, where the observed data is often corrupted by Poisson noise due to the nature of photon counting in imaging modalities such as \acrfull{pet} and \acrfull{spect}.
This algorithm was first proposed by \cite{Shepp1982} for PET image reconstruction and has since been widely used in various applications.
Let now consider the following degradation model -- that is for instance suited for PET imaging -- for the observed data $Y$:
\begin{equation}
    Y | X=x \sim \text{Poisson}(A x + b)
\end{equation}
where $A \in \mathbb{R}^{m \times n}$ is the system matrix, $b \in \mathbb{R}^m$ is the background noise, and $X \in \mathbb{R}^n$ is the unknown image to be estimated.
In emmission tomography, a good choice for the latent variable $Z$ is the number of photons emitted from each voxel and detected by each detector, which can be modeled as a Poisson random variable.
$Z_{ij}$ represents the number of photons emitted from voxel $j$ and detected by detector $i$, and is assumed to follow a Poisson distribution with mean $A_{ij} x_j$:
\begin{equation}
    Z_{ij} | X=x \sim \text{Poisson}(A_{ij} x_j) \
    \forall i Y_i = \sum_{j=1}^{n} Z_{ij}
\end{equation}
Plugging this choice of latent variable into the \acrshort{em} framework leads to the following \textit{complete-data likelihood}:
\begin{equation}
    p_{Y, Z}(y, z|x) = \prod_{i=1}^{m} \prod_{j=1}^{n} \frac{(A_{ij} x_j)^{z_{ij}} e^{-A_{ij} x_j}}{z_{ij}!}
\end{equation}
which we take the logarithm of to obtain the \textit{complete-data log-likelihood}:
\begin{equation}
    \log p_{Y, Z}(y, z|x) = \sum_{i=1}^{m} \sum_{j=1}^{n} z_{ij} \log(A_{ij} x_j) - A_{ij} x_j - \log(z_{ij}!)
\end{equation}
Now we can compute the tangent-majorant $Q$ by taking the expectation of the complete-data log-likelihood with respect to the conditional distribution of $Z$ given $Y$ and the current estimate $x_k$:
\begin{align}
    Q(x, x_k) &= \mathbb{E}_{Z|Y, x_k}[\log p_{Y, Z}(Y, Z|x)] \\
    &= \sum_{i=1}^{m} \sum_{j=1}^{n} \mathbb{E}_{Z|Y=y, x_k}[Z_{ij}] \log(A_{ij} x_j) - A_{ij} x_j + \mathbb{E}_{Z|Y=y, x_k}[\log(Z_{ij}!)]
\end{align}
$\mathbb{E}_{Z|Y, x_k}[\log(Z_{ij}!)]$ being independant of $x$, it can be dropped from the optimization problem and we just have to compute the expectation of $Z_{ij}$ given $Y$ and the current estimate $x_k$:
$Z$ being conditionned on the sum of its components $Y_i = \sum_{j=1}^{n} Z_{ij}$, it follows a multinomial distribution and we have:
\begin{equation}
    \mathbb{E}_{Z|Y=y, x_k}[Z_{ij}] = y_i \frac{A_{ij} x_{k,j}}{\sum_{l=1}^{n} A_{il} x_{k,l}} = y_i \frac{A_{ij} x_{k,j}}{(Ax_k)_i}
\end{equation}
which we can now plug into the optimization of the tangent-majorant $Q$ to obtain:
\begin{equation}
    Q(x, x_k) = \sum_{i=1}^{m} \sum_{j=1}^{n} y_i \frac{A_{ij} x_{k,j}}{(Ax_k)_i} \log(A_{ij} x_j) - A_{ij} x_j + \text{const}
\end{equation}
We take the derivative of $Q$ with respect to each component $x_j$ and set it to zero to find the optimal $x_j$:
\begin{align}
    \frac{\partial Q}{\partial x_j} &= \sum_{i=1}^{m} y_i \frac{A_{ij} x_{k,j}}{(Ax_k)_i} \frac{1}{x_j} - \sum_{i=1}^{m} A_{ij} = 0 \\
    x_j &= \frac{\sum_{i=1}^{m} y_i \frac{A_{ij} x_{k,j}}{(Ax_k)_i}}{\sum_{i=1}^{m} A_{ij}}
\end{align}
Setting $x_{k+1,j} = x_j$ gives the update rule for the \acrshort{mlem} algorithm:
\begin{equation}
    x_{k+1,j} = \frac{x_{k,j}}{\sum_{i=1}^{m} A_{ij}} \sum_{i=1}^{m} y_i \frac{A_{ij}}{(Ax_k)_i}
\end{equation}
The \acrshort{mlem} algorithm is summarized in Algorithm \ref{alg:mlem}.
\begin{algorithm}
    \caption{Maximum-Likelihood Expectation-Maximization (MLEM)}
    \begin{algorithmic}[1]
    \State \textbf{Input:} Measured data $y$, system matrix $A$, initial estimate $x^{(0)}$
    \State \textbf{Output:} Estimated image $x^{(k)}$
    \For{$k = 0$ to $K-1$}
        \State \textbf{E-step:} Compute the expected number of counts
        \begin{equation*}
            \hat{y}_i^{(k)} = (Ax^{(k)})_i
        \end{equation*}
        \State \textbf{M-step:} Update the image estimate
        \begin{equation*}
            x_j^{(k+1)} = x_j^{(k)} \frac{\sum_{i=1}^{d} A_{ij} \frac{y_i}{\hat{y}_i^{(k)}}}{\sum_{i=1}^{d} A_{ij}}
        \end{equation*}
    \EndFor
    \State \textbf{Return} $x^{(K)}$
    \end{algorithmic}
    \label{alg:mlem}
\end{algorithm}
This algorithm can be speed up by using the \textit{ordered subsets} technique introduced by \cite{Hudson1994}, which consists in dividing the data into subsets and updating the parameters using each subset in turn. This leads to the \acrfull{osem} algorithm, which is widely used in PET and SPECT image reconstruction.
Even if the algorithm is not designed to minimized a well-defined cost funcion, the latter converges to an MLEM image in the case of noiseless data, but this might not be true in practice and the solution might not even converge but cycle between similar but distinct images \cite{6152654}.
However, it is faster to get quality results compared to the MLEM algorithm and can be used in practice with a large number of subsets without significant loss of image quality.
\begin{algorithm}
    \caption{Ordered Subset Expectation Maximization (OSEM)}
    \begin{algorithmic}[1]
    \State \textbf{Input:} Measured data $y$, system matrix $A$, initial estimate $x^{(0)}$, number of subsets $Q$
    \State \textbf{Output:} Estimated image $x^{(k)}$
    \For{$k = 0$ to $K-1$}
        \For{$q = 1$ to $Q$}
            \State \textbf{E-step:} Compute the expected number of counts for subset $q$
            \begin{equation*}
                \hat{y}_{i}^{(k,q)} = (Ax^{(k)})_i \quad \text{for } i \in \text{subset } q
            \end{equation*}
            \State \textbf{M-step:} Update the image estimate using subset $q$
            \begin{equation*}
                x_j^{(k,q)} = x_j^{(k,q-1)} \frac{\sum_{i \in \text{subset } q} A_{ij} \frac
                {y_i}{\hat{y}_i^{(k,q)}}}{\sum_{i \in \text{subset } q} A_{ij}}
            \end{equation*}
        \EndFor
    \EndFor
    \State \textbf{Return} $x^{(K)}$
    \end{algorithmic}
    \label{alg:osem}
\end{algorithm}

\acrfull{mlem} and \acrfull{osem} are particularly sensitive to the noise in the data and the fact that the system matrix is ill-conditionned and can produce images with high variance if the number of iterations is too large.
Hence, post-processing techniques such as gaussian filtering or denoising techniques are often applied to the reconstructed images to reduce the noise and improve the image quality.
Another way to tackle this problem is to use regularization techniques. The most common approach with \acrshort{mlem} and \acrshort{osem} is to stop the iterations early, which is known as \textit{early stopping}. This can be seen as a form of regularization, as it prevents the algorithm from overfitting to the noise in the data. % TODO add plots
These algorithms can also be modified to include a regularization term incorporing prior knowledge in the optimization problem, which can help to reduce the noise and improve the image quality.
